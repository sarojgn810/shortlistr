  {\large\bfseries #1}%
}

  {\fontsize{23}{26}\selectfont\bfseries #1}\par\vspace{3pt}%
  {\small\color{cvmuted} #2}\par\vspace{5pt}%
  {\color{cvaccent}\hrule height 2pt}%
  \vspace{2pt}%
}
